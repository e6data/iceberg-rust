18//! Parquet-based manifest serialization for columnar metadata projection.
19//!
20//! Replaces Avro manifest files with Parquet to enable the executor to read
21//! only the manifest columns needed for query planning (e.g., file_path +
22//! partition bounds) without deserializing all 200+ column statistics.
23//!
24//! Same v2 semantics as Avro manifests — just a different serialization format.
25//!
26//! Map-typed columns (column_sizes, lower_bounds, upper_bounds, etc.) are
27//! stored as JSON-encoded binary blobs. This is simpler than Arrow Map types
28//! and still allows projection to skip these columns entirely when not needed.
29

56/// Build a flat Arrow schema for manifest entries.
57///
58/// Map-typed columns use Binary (JSON-encoded) for simplicity.
59/// This still allows columnar projection — the executor skips map columns
60/// when it only needs file_path + partition + status.
61pub fn manifest_arrow_schema() -> ArrowSchema {
62    ArrowSchema::new(vec![
63        Field::new("status", DataType::Int32, false),
64        Field::new("snapshot_id", DataType::Int64, true),
65        Field::new("sequence_number", DataType::Int64, true),
66        Field::new("file_sequence_number", DataType::Int64, true),
67        Field::new("content", DataType::Int32, false),
68        Field::new("file_path", DataType::Utf8, false),
69        Field::new("file_format", DataType::Utf8, false),
70        Field::new("partition_json", DataType::Utf8, true),
71        Field::new("record_count", DataType::Int64, false),
72        Field::new("file_size_in_bytes", DataType::Int64, false),
73        // Map columns as JSON binary — allows projection to skip them
74        Field::new("column_sizes_json", DataType::Binary, true),
75        Field::new("value_counts_json", DataType::Binary, true),
76        Field::new("null_value_counts_json", DataType::Binary, true),
77        Field::new("nan_value_counts_json", DataType::Binary, true),
78        Field::new("lower_bounds_json", DataType::Binary, true),
79        Field::new("upper_bounds_json", DataType::Binary, true),
80        Field::new("key_metadata", DataType::Binary, true),
81        Field::new("split_offsets_json", DataType::Binary, true),
82        Field::new("equality_ids_json", DataType::Binary, true),
83        Field::new("sort_order_id", DataType::Int32, true),
84        Field::new("partition_spec_id", DataType::Int32, false),
85    ])
86}

92/// Encode manifest metadata as HashMap for Arrow schema metadata.
93/// This gets written as Parquet file-level key-value metadata.
94pub fn encode_manifest_metadata(metadata: &ManifestMetadata) -> HashMap<String, String> {
95    let mut kv = HashMap::new();
96    kv.insert("schema".to_string(), serde_json::to_string(metadata.schema.as_ref()).unwrap_or_default());
97    kv.insert("schema-id".to_string(), metadata.schema_id.to_string());
98    kv.insert("partition-spec".to_string(), serde_json::to_string(&metadata.partition_spec.fields()).unwrap_or_default());
99    kv.insert("partition-spec-id".to_string(), metadata.partition_spec.spec_id().to_string());
100    kv.insert("format-version".to_string(), (metadata.format_version as u8).to_string());
101    kv.insert("content".to_string(), metadata.content.to_string());
102    kv
103}

109/// Write manifest entries to a Parquet-format byte buffer.
110pub fn write_parquet_manifest(
111    entries: &[ManifestEntry],
112    metadata: &ManifestMetadata,
113    partition_type: &StructType,
114) -> Result<Vec<u8>> {
115    // Embed manifest metadata in the Arrow schema's metadata field.
116    // ArrowWriter propagates this to the Parquet file-level key-value metadata.
117    let kv_metadata = encode_manifest_metadata(metadata);
118    let schema = Arc::new(manifest_arrow_schema().with_metadata(kv_metadata));
119    let batch = manifest_entries_to_record_batch(entries, &schema, partition_type, metadata.format_version)?;
120
121    let props = WriterProperties::builder()
122        .set_compression(Compression::ZSTD(Default::default()))
123        .build();
124
125    let mut buf = Vec::new();
126    let mut writer = ArrowWriter::try_new(&mut buf, schema, Some(props))
127        .map_err(|e| Error::new(ErrorKind::Unexpected, format!("Failed to create parquet writer: {e}")))?;
128
129    writer.write(&batch)
130        .map_err(|e| Error::new(ErrorKind::Unexpected, format!("Failed to write batch: {e}")))?;
131    writer.close()
132        .map_err(|e| Error::new(ErrorKind::Unexpected, format!("Failed to close writer: {e}")))?;
133
134    Ok(buf)
135}

137/// Convert ManifestEntry slice to Arrow RecordBatch.
138fn manifest_entries_to_record_batch(
139    entries: &[ManifestEntry],
140    schema: &Arc<ArrowSchema>,
141    partition_type: &StructType,
142    format_version: FormatVersion,
143) -> Result<RecordBatch> {
144    let n = entries.len();
145
146    let mut status = Int32Builder::with_capacity(n);
147    let mut snapshot_id = Int64Builder::with_capacity(n);
148    let mut seq_num = Int64Builder::with_capacity(n);
149    let mut file_seq = Int64Builder::with_capacity(n);
150    let mut content = Int32Builder::with_capacity(n);
151    let mut file_path = StringBuilder::with_capacity(n, n * 100);
152    let mut file_format = StringBuilder::with_capacity(n, n * 8);
153    let mut partition_json = StringBuilder::with_capacity(n, n * 64);
154    let mut record_count = Int64Builder::with_capacity(n);
155    let mut file_size = Int64Builder::with_capacity(n);
156    let mut column_sizes_json = BinaryBuilder::with_capacity(n, n * 128);
157    let mut value_counts_json = BinaryBuilder::with_capacity(n, n * 128);
158    let mut null_value_counts_json = BinaryBuilder::with_capacity(n, n * 128);
159    let mut nan_value_counts_json = BinaryBuilder::with_capacity(n, n * 64);
160    let mut lower_bounds_json = BinaryBuilder::with_capacity(n, n * 256);
161    let mut upper_bounds_json = BinaryBuilder::with_capacity(n, n * 256);
162    let mut key_metadata_b = BinaryBuilder::with_capacity(n, n * 16);
163    let mut split_offsets_json = BinaryBuilder::with_capacity(n, n * 64);
164    let mut equality_ids_json = BinaryBuilder::with_capacity(n, n * 32);
165    let mut sort_order_id = Int32Builder::with_capacity(n);
166    let mut part_spec_id = Int32Builder::with_capacity(n);
167
168    for entry in entries {
169        let df = &entry.data_file;
170
171        status.append_value(entry.status as i32);
172        match entry.snapshot_id {
173            Some(v) => snapshot_id.append_value(v),
174            None => snapshot_id.append_null(),
175        }
176        match entry.sequence_number {
177            Some(v) => seq_num.append_value(v),
178            None => seq_num.append_null(),
179        }
180        match entry.file_sequence_number {
181            Some(v) => file_seq.append_value(v),
182            None => file_seq.append_null(),
183        }
184        content.append_value(df.content as i32);
185        file_path.append_value(&df.file_path);
186        file_format.append_value(df.file_format.to_string().to_ascii_uppercase());
187
188        // Partition as JSON string
189        let part_str = serialize_partition_json(&df.partition, partition_type);
190        partition_json.append_value(&part_str);
191
192        record_count.append_value(df.record_count as i64);
193        file_size.append_value(df.file_size_in_bytes as i64);
194
195        // Map columns as JSON bytes
196        append_map_json(&mut column_sizes_json, &serialize_i64_map(&df.column_sizes));
197        append_map_json(&mut value_counts_json, &serialize_i64_map(&df.value_counts));
198        append_map_json(&mut null_value_counts_json, &serialize_i64_map(&df.null_value_counts));
199        append_map_json(&mut nan_value_counts_json, &serialize_i64_map(&df.nan_value_counts));
200
201        // Bounds: serialize field_id → base64(bytes) as JSON
202        append_map_json(&mut lower_bounds_json, &serialize_bounds_map(&df.lower_bounds));
203        append_map_json(&mut upper_bounds_json, &serialize_bounds_map(&df.upper_bounds));
204
205        match &df.key_metadata {
206            Some(km) => key_metadata_b.append_value(km.as_slice()),
207            None => key_metadata_b.append_null(),
208        }
209        append_opt_json(&mut split_offsets_json, &df.split_offsets);
210        append_opt_json(&mut equality_ids_json, &df.equality_ids);
211        match df.sort_order_id {
212            Some(v) => sort_order_id.append_value(v),
213            None => sort_order_id.append_null(),
214        }
215        part_spec_id.append_value(df.partition_spec_id);
216    }
217
218    let columns: Vec<ArrayRef> = vec![
219        Arc::new(status.finish()),
220        Arc::new(snapshot_id.finish()),
221        Arc::new(seq_num.finish()),
222        Arc::new(file_seq.finish()),
223        Arc::new(content.finish()),
224        Arc::new(file_path.finish()),
225        Arc::new(file_format.finish()),
226        Arc::new(partition_json.finish()),
227        Arc::new(record_count.finish()),
228        Arc::new(file_size.finish()),
229        Arc::new(column_sizes_json.finish()),
230        Arc::new(value_counts_json.finish()),
231        Arc::new(null_value_counts_json.finish()),
232        Arc::new(nan_value_counts_json.finish()),
233        Arc::new(lower_bounds_json.finish()),
234        Arc::new(upper_bounds_json.finish()),
235        Arc::new(key_metadata_b.finish()),
236        Arc::new(split_offsets_json.finish()),
237        Arc::new(equality_ids_json.finish()),
238        Arc::new(sort_order_id.finish()),
239        Arc::new(part_spec_id.finish()),
240    ];
241
242    RecordBatch::try_new(schema.clone(), columns)
243        .map_err(|e| Error::new(ErrorKind::Unexpected, format!("Failed to build RecordBatch: {e}")))
244}

246fn serialize_partition_json(partition: &Struct, partition_type: &StructType) -> String {
247    // Use RawLiteral for proper Iceberg partition value serialization.
248    // RawLiteral handles all type conversions (timestamps, decimals, etc.) correctly.
249    match RawLiteral::try_from(Literal::Struct(partition.clone()), &Type::Struct(partition_type.clone())) {
250        Ok(raw) => serde_json::to_string(&raw).unwrap_or_default(),
251        Err(_) => "null".to_string(),
252    }
253}
254
255fn serialize_i64_map(m: &HashMap<i32, u64>) -> String {
256    if m.is_empty() {
257        return String::new();
258    }
259    // Serialize as JSON: {"field_id": value, ...}
260    let map: HashMap<String, u64> = m.iter().map(|(k, v)| (k.to_string(), *v)).collect();
261    serde_json::to_string(&map).unwrap_or_default()
262}
263
264fn serialize_bounds_map(m: &HashMap<i32, Datum>) -> String {
265    if m.is_empty() {
266        return String::new();
267    }
268    // Serialize as JSON: {"field_id": "base64(bytes)", ...}
269    use base64::Engine;
270    let map: HashMap<String, String> = m
271        .iter()
272        .filter_map(|(k, v)| {
273            v.to_bytes().ok().map(|bytes| {
274                (k.to_string(), base64::engine::general_purpose::STANDARD.encode(bytes.as_ref()))
275            })
276        })
277        .collect();
278    serde_json::to_string(&map).unwrap_or_default()
279}
280
281fn append_map_json(builder: &mut BinaryBuilder, json: &str) {
282    if json.is_empty() {
283        builder.append_null();
284    } else {
285        builder.append_value(json.as_bytes());
286    }
287}
288
289fn append_opt_json<T: serde::Serialize>(builder: &mut BinaryBuilder, val: &Option<T>) {
290    match val {
291        Some(v) => {
292            let json = serde_json::to_string(v).unwrap_or_default();
293            builder.append_value(json.as_bytes());
294        }
295        None => builder.append_null(),
296    }
297}

303/// Read manifest entries from a Parquet-format manifest file.
304pub fn read_parquet_manifest(bytes: &[u8]) -> Result<(ManifestMetadata, Vec<ManifestEntry>)> {
305    let reader = ParquetRecordBatchReaderBuilder::try_new(Bytes::copy_from_slice(bytes))
306        .map_err(|e| Error::new(ErrorKind::DataInvalid, format!("Failed to open parquet manifest: {e}")))?;
307
308    // ArrowWriter stores Arrow schema metadata in the Parquet file-level key-value metadata
309    // under the key "ARROW:schema". Our manifest metadata keys are embedded alongside it.
310    // We also check the Arrow schema metadata directly.
311    let parquet_metadata = reader.metadata().clone();
312    let arrow_schema = reader.schema();
313
314    // First try: Arrow schema metadata (ArrowWriter propagates schema.metadata here)
315    let arrow_meta = arrow_schema.metadata();
316    let metadata = if arrow_meta.contains_key("schema") {
317        let map: HashMap<String, Vec<u8>> = arrow_meta
318            .iter()
319            .map(|(k, v)| (k.clone(), v.as_bytes().to_vec()))
320            .collect();
321        ManifestMetadata::parse(&map)?
322    } else {
323        // Fallback: Parquet file-level key-value metadata
324        let kv_metadata = parquet_metadata.file_metadata().key_value_metadata();
325        parse_parquet_manifest_metadata(kv_metadata)?
326    };
327
328    let partition_type = metadata.partition_spec.partition_type(&metadata.schema)?;
329
330    let batch_reader = reader
331        .build()
332        .map_err(|e| Error::new(ErrorKind::DataInvalid, format!("Failed to build reader: {e}")))?;
333
334    let mut entries = Vec::new();
335    for batch_result in batch_reader {
336        let batch = batch_result
337            .map_err(|e| Error::new(ErrorKind::DataInvalid, format!("Failed to read batch: {e}")))?;
338        let batch_entries = record_batch_to_manifest_entries(
339            &batch,
340            &metadata,
341            &partition_type,
342        )?;
343        entries.extend(batch_entries);
344    }
345
346    Ok((metadata, entries))
347}

349/// Read manifest entries with column projection.
350///
351/// Only the specified columns are read from the Parquet file. Columns not
352/// requested will be empty/default in the returned ManifestEntry.
353///
354/// Common projection sets:
355/// - Planning: `["status", "content", "file_path", "file_format", "record_count", "file_size_in_bytes", "partition_spec_id"]`
356/// - Pruning:  above + `["lower_bounds_json", "upper_bounds_json"]`
357/// - Full:     all columns (use `read_parquet_manifest` instead)
358pub fn read_parquet_manifest_projected(
359    bytes: &[u8],
360    columns: &[&str],
361) -> Result<(ManifestMetadata, Vec<ManifestEntry>)> {
362    let builder = ParquetRecordBatchReaderBuilder::try_new(Bytes::copy_from_slice(bytes))
363        .map_err(|e| Error::new(ErrorKind::DataInvalid, format!("Failed to open parquet manifest: {e}")))?;
364
365    let arrow_schema = builder.schema();
366    let arrow_meta = arrow_schema.metadata();
367    let metadata = if arrow_meta.contains_key("schema") {
368        let map: HashMap<String, Vec<u8>> = arrow_meta
369            .iter()
370            .map(|(k, v)| (k.clone(), v.as_bytes().to_vec()))
371            .collect();
372        ManifestMetadata::parse(&map)?
373    } else {
374        let parquet_metadata = builder.metadata().clone();
375        let kv_metadata = parquet_metadata.file_metadata().key_value_metadata();
376        parse_parquet_manifest_metadata(kv_metadata)?
377    };
378
379    // Build projection mask from requested columns
380    let parquet_schema = builder.parquet_schema();
381    let mut projection_indices = Vec::new();
382    for col_name in columns {
383        for (idx, field) in arrow_schema.fields().iter().enumerate() {
384            if field.name() == *col_name {
385                projection_indices.push(idx);
386                break;
387            }
388        }
389    }
390
391    let projection = parquet::arrow::ProjectionMask::leaves(
392        parquet_schema,
393        projection_indices,
394    );
395
396    let partition_type = metadata.partition_spec.partition_type(&metadata.schema)?;
397
398    let batch_reader = builder
399        .with_projection(projection)
400        .build()
401        .map_err(|e| Error::new(ErrorKind::DataInvalid, format!("Failed to build projected reader: {e}")))?;
402
403    let mut entries = Vec::new();
404    for batch_result in batch_reader {
405        let batch = batch_result
406            .map_err(|e| Error::new(ErrorKind::DataInvalid, format!("Failed to read batch: {e}")))?;
407        let batch_entries = record_batch_to_manifest_entries(&batch, &metadata, &partition_type)?;
408        entries.extend(batch_entries);
409    }
410
411    Ok((metadata, entries))
412}

414fn parse_parquet_manifest_metadata(
415    kv: Option<&Vec<parquet::file::metadata::KeyValue>>,
416) -> Result<ManifestMetadata> {
417    let kv = kv.ok_or_else(|| {
418        Error::new(ErrorKind::DataInvalid, "Parquet manifest missing key-value metadata")
419    })?;
420
421    let mut map: HashMap<String, Vec<u8>> = HashMap::new();
422    for entry in kv {
423        if let Some(ref v) = entry.value {
424            map.insert(entry.key.clone(), v.as_bytes().to_vec());
425        }
426    }
427
428    ManifestMetadata::parse(&map)
429}

431fn record_batch_to_manifest_entries(
432    batch: &RecordBatch,
433    metadata: &ManifestMetadata,
434    partition_type: &StructType,
435) -> Result<Vec<ManifestEntry>> {
436    let n = batch.num_rows();
437    let mut entries = Vec::with_capacity(n);
438
439    let status_arr = col_i32(batch, "status")?;
440    let snapshot_id_arr = col_i64_opt(batch, "snapshot_id");
441    let seq_num_arr = col_i64_opt(batch, "sequence_number");
442    let file_seq_arr = col_i64_opt(batch, "file_sequence_number");
443    let content_arr = col_i32(batch, "content")?;
444    let file_path_arr = col_str(batch, "file_path")?;
445    let file_format_arr = col_str(batch, "file_format")?;
446    let partition_json_arr = col_str_opt(batch, "partition_json");
447    let record_count_arr = col_i64(batch, "record_count")?;
448    let file_size_arr = col_i64(batch, "file_size_in_bytes")?;
449    let column_sizes_arr = col_binary_opt(batch, "column_sizes_json");
450    let value_counts_arr = col_binary_opt(batch, "value_counts_json");
451    let null_value_counts_arr = col_binary_opt(batch, "null_value_counts_json");
452    let nan_value_counts_arr = col_binary_opt(batch, "nan_value_counts_json");
453    let lower_bounds_arr = col_binary_opt(batch, "lower_bounds_json");
454    let upper_bounds_arr = col_binary_opt(batch, "upper_bounds_json");
455    let key_metadata_arr = col_binary_opt(batch, "key_metadata");
456    let split_offsets_arr = col_binary_opt(batch, "split_offsets_json");
457    let equality_ids_arr = col_binary_opt(batch, "equality_ids_json");
458    let sort_order_id_arr = col_i32_opt(batch, "sort_order_id");
459    let part_spec_id_arr = col_i32_opt(batch, "partition_spec_id");
460
461    for i in 0..n {
462        let status: ManifestStatus = status_arr.value(i).try_into()?;
463        let snapshot_id = nullable_i64(snapshot_id_arr, i);
464        let sequence_number = nullable_i64(seq_num_arr, i);
465        let file_sequence_number = nullable_i64(file_seq_arr, i);
466
467        let content_type: DataContentType = content_arr.value(i).try_into()?;
468        let file_path = file_path_arr.value(i).to_string();
469        let file_format: DataFileFormat = file_format_arr.value(i).parse()?;
470        let record_count = record_count_arr.value(i) as u64;
471        let file_size = file_size_arr.value(i) as u64;
472
473        let partition = parse_partition_json(
474            partition_json_arr.and_then(|a| if Array::is_null(a, i) { None } else { Some(a.value(i)) }),
475            partition_type,
476        );
477
478        let column_sizes = parse_i64_map_json(read_binary_opt(column_sizes_arr, i));
479        let value_counts = parse_i64_map_json(read_binary_opt(value_counts_arr, i));
480        let null_value_counts = parse_i64_map_json(read_binary_opt(null_value_counts_arr, i));
481        let nan_value_counts = parse_i64_map_json(read_binary_opt(nan_value_counts_arr, i));
482
483        let lower_bounds = parse_bounds_map_json(read_binary_opt(lower_bounds_arr, i), &metadata.schema);
484        let upper_bounds = parse_bounds_map_json(read_binary_opt(upper_bounds_arr, i), &metadata.schema);
485
486        let key_metadata_val = read_binary_opt(key_metadata_arr, i).map(|b| b.to_vec());
487        let split_offsets: Option<Vec<i64>> = read_binary_opt(split_offsets_arr, i)
488            .and_then(|b| serde_json::from_slice(b).ok());
489        let equality_ids: Option<Vec<i32>> = read_binary_opt(equality_ids_arr, i)
490            .and_then(|b| serde_json::from_slice(b).ok());
491        let sort_id = sort_order_id_arr.and_then(|a| if Array::is_null(a, i) { None } else { Some(a.value(i)) });
492        let spec_id = part_spec_id_arr.map(|a| a.value(i)).unwrap_or(metadata.partition_spec.spec_id());
493
494        let data_file = DataFile {
495            content: content_type,
496            file_path,
497            file_format,
498            partition,
499            record_count,
500            file_size_in_bytes: file_size,
501            column_sizes,
502            value_counts,
503            null_value_counts,
504            nan_value_counts,
505            lower_bounds,
506            upper_bounds,
507            key_metadata: key_metadata_val,
508            split_offsets,
509            equality_ids,
510            sort_order_id: sort_id,
511            partition_spec_id: spec_id,
512            first_row_id: None,
513            referenced_data_file: None,
514            content_offset: None,
515            content_size_in_bytes: None,
516        };
517
518        entries.push(ManifestEntry {
519            status,
520            snapshot_id,
521            sequence_number,
522            file_sequence_number,
523            data_file,
524        });
525    }
526
527    Ok(entries)
528}

534fn parse_partition_json(json: Option<&str>, partition_type: &StructType) -> Struct {
535    let Some(json_str) = json else { return Struct::empty() };
536    if json_str == "null" || json_str.is_empty() {
537        return Struct::empty();
538    }
539
540    // Deserialize via RawLiteral for proper Iceberg type handling
541    let raw: RawLiteral = match serde_json::from_str(json_str) {
542        Ok(r) => r,
543        Err(_) => return Struct::empty(),
544    };
545
546    match raw.try_into(&Type::Struct(partition_type.clone())) {
547        Ok(Some(Literal::Struct(s))) => s,
548        _ => Struct::empty(),
549    }
550}

552fn parse_i64_map_json(bytes: Option<&[u8]>) -> HashMap<i32, u64> {
553    let Some(b) = bytes else { return HashMap::new() };
554    let Ok(map) = serde_json::from_slice::<HashMap<String, u64>>(b) else {
555        return HashMap::new();
556    };
557    map.into_iter()
558        .filter_map(|(k, v)| k.parse::<i32>().ok().map(|k| (k, v)))
559        .collect()
560}
561
562fn parse_bounds_map_json(bytes: Option<&[u8]>, schema: &Schema) -> HashMap<i32, Datum> {
563    let Some(b) = bytes else { return HashMap::new() };
564    let Ok(map) = serde_json::from_slice::<HashMap<String, String>>(b) else {
565        return HashMap::new();
566    };
567    use base64::Engine;
568    let mut result = HashMap::new();
569    for (k_str, v_b64) in map {
570        let Ok(field_id) = k_str.parse::<i32>() else { continue };
571        let Ok(raw_bytes) = base64::engine::general_purpose::STANDARD.decode(&v_b64) else { continue };
572        // Find the field type from schema to reconstruct the Datum
573        if let Some(field) = schema.field_by_id(field_id) {
574            if let Some(prim_type) = field.field_type.as_primitive_type() {
575                if let Ok(datum) = Datum::try_from_bytes(&raw_bytes, prim_type.clone()) {
576                    result.insert(field_id, datum);
577                }
578            }
579        }
580    }
581    result
582}
